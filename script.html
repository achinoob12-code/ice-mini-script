local function gmc()
    local c=game.Players.LocalPlayer.Character if not c then return end
    local h=c:FindFirstChildOfClass("Humanoid")if not h or not h.SeatPart then return end
    return h.SeatPart.Parent:FindFirstChild("Body")or h.SeatPart.Parent.PrimaryPart or h.SeatPart.Parent:FindFirstChildWhichIsA("BasePart")
end

bf.MouseButton1Click:Connect(function()
    T.f=not T.f
    T.F,T.B,T.L,T.R,T.U,T.D=false,false,false,false,false,false
    if T.f then
        local p=gmc()
        if p then
            workspace.Gravity=0
            p.CustomPhysicalProperties=PhysicalProperties.new(0,0,0,0,0)
            bf.Text="ปิดโหมดบิน"
            bf.BackgroundColor3=Color3.new(.7,.2,.2)
        end
    else
        local p=gmc()
        if p then
            workspace.Gravity=196.2
            p.CustomPhysicalProperties=PhysicalProperties.new(0.7,100,0.5,1,1)
            bf.Text="เปิดโหมดบิน"
            bf.BackgroundColor3=Color3.new(.2,.7,.3)
        end
    end
end)

game:GetService("RunService").RenderStepped:Connect(function()
    if not T.f then return end
    local p=gmc()if not p then return end
    local cf=p.CFrame
    local mv=Vector3.new(
        (T.L and -1 or 0)+(T.R and 1 or 0),
        (T.U and 1 or 0)+(T.D and -1 or 0),
        (T.F and -1 or 0)+(T.B and 1 or 0)
    )
    if mv.Magnitude>0 then
        mv=mv.Unit*T.v
        local rot=CFrame.Angles(0,math.rad((T.L and 1 or 0)+(T.R and -1 or 0))*90*T.t,0)
        p.Velocity=cf:VectorToWorldSpace(mv)
        p.CFrame=cf*rot
    else
        p.Velocity=Vector3.new(0,0,0)
    end
end)
